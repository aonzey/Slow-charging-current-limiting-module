#  单位统一「微安 μA」：1A = 1000000（别少写三个零）
#
#  v3 修复（基于实测诊断）：
#    - battery_charging_enabled 在本机是"假开关"（能写但内核不认），已弃用
#    - 改用 input_suspend（反极性：0=充电 / 1=停充），实测有效
#    - 停充后 status 会变 Discharging，守护逻辑已改为不依赖 status 判断
# ============================================================

# ---------- 1. 出厂电流上限（安全边界）----------
# 实测出厂值 = 4920000 μA (4.92A)
# 模块绝不会写入超过此值的电流
FACTORY_MAX=4920000

# ---------- 2. 电池端最大充电电流（主要降温手段）----------
# 可在 0 ~ FACTORY_MAX 之间自由调整
#
#   3000000 = 3.0A  出厂的 61%，实测 30-34℃  ← 当前值（四轮验证通过）
#   2000000 = 2.0A  出厂的 41%，实测 27-31℃
#   1500000 = 1.5A  出厂的 31%
#   4000000 = 4.0A  出厂的 81%，可能回到 36℃+，需配合提高 USB_UA
CURRENT_UA=3000000

# ---------- 3. USB 输入端最大电流 ----------
# 填 0 = 不限制输入端
# 本机真实协商值在 usb/pd_current_max 上（约 1550000）
# 此项保持"只降不升"策略（不会抬高输入上限）
USB_UA=1500000

# ---------- 4. 按电量停充 ----------
# 填 0 = 关闭此功能
# 80 = 充到 80% 停止，掉到 RESUME_AT 再恢复
# 实测 input_suspend=1 时电池转为放电（约 +230mA），
# 所以会缓慢从 80% 放到 70% 再补回，形成循环——这是迟滞设计的预期行为
# 若嫌循环频繁，把 RESUME_AT 调小（如 60）可减少循环次数
CHARGE_STOP=80
RESUME_AT=70

# ---------- 5. 守护循环 ----------
# 1 = 开启（每 INTERVAL 秒检查，拔掉充电器时也保持低功耗轮询）
# 0 = 关闭（只在开机时写一次，停充功能将不可用）
DAEMON=1

# 守护检查间隔（秒），不宜小于 15
INTERVAL=30

# 开机后延迟多少秒再写入（等充电驱动就绪）
DELAY=20
