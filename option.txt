# ============================================================
#  Slow Charge 限流慢充 v4 —— alioth (Redmi K40)
#  改完无需重启，执行：
#    /data/adb/ap/bin/busybox sh /data/adb/modules/slowcharge/service.sh
#
#  单位统一「微安 μA」：1A = 1000000（别少写三个零）
#
#  ============ v4 核心改动（针对"睡醒变 100%"问题）============
#
#  【旧方案为什么失效】
#    v3 用 input_suspend=1 硬停充 -> battery/status 变成 Discharging
#    -> Android 判定"未充电" -> 进入 Doze 深度休眠
#    -> 脚本里的 sleep 30 不再推进 -> 守护被冻结数小时（实测 14.5 小时无日志）
#    -> 冻结期间没人监督，input_suspend 被系统清掉 -> 一路充到 100%
#
#  【v4 新方案：软停充（trickle）】
#    到阈值时不是断开充电，而是把电流降到极小值（默认 150mA）。
#    status 始终保持 Charging -> Android 判定"正在充电" -> 不进 Doze
#    -> 守护永不冻结，从根本上解决问题，且不需要 wakelock（零额外耗电）
#
#    涓流值必须接近待机耗电（约 50mA），否则电量仍会持续上涨：
#      150mA -> 净充 100mA -> 每小时约 2.2%，一夜涨约 17.7%，最终仍会满
#      60mA  -> 净充 10mA  -> 每小时约 0.22%，一夜仅涨约 1.8%（推荐）
#    4520mAh 电池，1% ≈ 45.2mAh
#
#  【实测依据】
#    constant_charge_current_max 在四轮测试中从未被小米守护进程覆盖，
#    是本方案最可靠的节点 —— 软停充正是复用这个可靠性
# ============================================================

# ---------- 1. 出厂电流上限（安全边界）----------
# 实测出厂值 = 4920000 μA (4.92A)，模块绝不会写入超过此值
FACTORY_MAX=4920000

# ---------- 2. 正常充电时的电流（主要降温手段）----------
# 可在 0 ~ FACTORY_MAX 之间自由调整
#
#   3000000 = 3.0A  出厂的 61%，实测 30-34℃  ← 已通过四轮验证
#   2000000 = 2.0A  出厂的 41%，实测 27-31℃
#   1500000 = 1.5A  出厂的 31%
#   1000000 = 1.0A  出厂的 20%，约 4W，极凉
CURRENT_UA=3000000

# ---------- 3. 软停充时的涓流电流 ----------
# 到达 CHARGE_STOP 后，电流降到此值
#
# 【重要】净充速率 = TRICKLE_UA - 待机耗电（约 50000 uA = 50mA）
#   若设 150000：净充 100mA -> 每小时约 2.2%，8 小时涨约 17.7%
#                从 94% 起算约 2.7 小时就充满到 100%，无法阻止充满！
#   若设 60000： 净充 10mA  -> 每小时约 0.22%，8 小时仅涨约 1.8%
#                （推荐，电量基本维持）
#   若设 50000： 净充约 0，完全维持；但亮屏使用时可能掉电
#                （掉电也不用怕，迟滞逻辑会在 45% 恢复快充）
#
# K40 电池 4520mAh，1% ≈ 45.2mAh
TRICKLE_UA=60000

# ---------- 4. 停充模式 ----------
#   trickle = 软停充（默认，推荐）
#             保持 status=Charging，天然免疫 Doze，不需要 wakelock
#   suspend = 硬停充（v3 的方式）
#             用 input_suspend 真断开充电路径，但会让 status 变 Discharging
#             触发 Doze，必须配合 ANTI_DOZE=1 才能可靠
STOP_MODE=trickle

# ---------- 5. 按电量停充 ----------
# 填 0 = 关闭此功能
# 60 = 电量到 60% 进入涓流，掉到 45% 恢复 3A
#
# 为什么是 60/45 而不是 80/70（长期插电的备用机）：
#   锂电池老化主要由「日历老化（静置损耗）」主导，存放电量越高衰减越快
#   25℃ 存放一年：100% 电量损失约 20%，75% 损失约 12-15%，50% 仅损失约 4%
#   60/45 区间平均 SOC 约 52%，接近理论最优
CHARGE_STOP=60
RESUME_AT=45

# ---------- 6. 硬停充模式下的 Doze 防护 ----------
# 仅当 STOP_MODE=suspend 时有意义；STOP_MODE=trickle 时无需开启
#   0 = 关闭
#   1 = wakelock（停充期间持锁，阻止休眠）
#   2 = wakelock + dumpsys deviceidle disable（更彻底，代价更大）
ANTI_DOZE=0

# ---------- 7. USB 输入端最大电流 ----------
# 填 0 = 不限制输入端
# 本机真实协商值在 usb/pd_current_max（约 1550000），
# 当前 3A 方案下输入端与电池端几乎同时到顶，是共同瓶颈
# 此项保持「只降不升」策略（不会抬高输入上限）
USB_UA=1500000

# ---------- 8. 守护循环 ----------
# 1 = 开启（每 INTERVAL 秒检查）
# 0 = 关闭（只在开机时写一次，停充功能不可用）
DAEMON=1

# 守护检查间隔（秒），不宜小于 15
INTERVAL=30

# 开机后延迟多少秒再写入（等充电驱动就绪）
DELAY=20
